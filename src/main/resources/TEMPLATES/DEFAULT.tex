% --- Custom Commands for Spaces ---

% 1. \blank{width}: Creates an underlined blank space (e.g., for fill-in-the-blanks)
% Usage: \blank{3cm}
\newcommand{\blank}[1]{\underline{\hspace{#1}}}
\newcommand{\qMarks}[1]{%
\ifthenelse{#1>0}{
\hfill\textbf{(#1 Mark\ifthenelse{#1>1}{s}{})}
}{}%
}
% 2. \workarea{height}: Creates empty vertical space for showing work
% Usage: \workarea{5cm}
\newcommand{\workarea}[1]{\vspace{#1}}

% 3. \answerline: Creates a full-width horizontal line for writing answers
% Usage: \answerline
\newcommand{\answerline}{\noindent\rule{\textwidth}{0.4pt}}
\usepackage{pgffor} % Required for the loop

% Usage: \answerlines{5} will create 5 lines with space to write
\newcommand{\answerlines}[1]{%
    \foreach \n in {1,...,#1}{%
        \noindent\rule{\linewidth}{0.4pt}\\[0.8cm] % Adjust 0.8cm to change line height
    }%
}
% 4. \answerbox{height}: Creates a framed box of a specific height
% Usage: \answerbox{5cm}
\newcommand{\answerbox}[1]{%
\\
\\
    \noindent\framebox[\linewidth][c]{\rule{0pt}{#1}}%
}

\newcommand{\questionAnsBox}[3]{
#1 \qMarks{#2} \answerbox{#3}
}
\newcommand{\questionLines}[3]{
#1 \qMarks{#2} \\ \\  \answerlines{#3}
}
\newcommand{\questionItemBox}[3]{
\item
\questionAnsBoxl{#1}{#2}{#3}
}
\newcommand{\questionItemLine}[3]{
\item
\questionLines{#1}{#2}{#3}
}

\end{enumerate} #1
